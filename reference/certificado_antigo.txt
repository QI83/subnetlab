function renderResult(){

  const passed=S.lives>0;

  const topicHtml=renderTopicStats();
  const hasWrong=S.wrongLog.length>0;
  const actionBtns=`<div class="btn-row" style="margin-top:16px">
    ${hasWrong?`<button class="btn blue" id="reviewBtn" style="padding:11px 20px">🔍 Revisar ${S.wrongLog.length} erro${S.wrongLog.length>1?'s':''}</button>`:''}
    <button class="btn" id="histBtn2" style="padding:11px 20px">📊 Histórico</button>
    <button class="btn" id="shareBtn" style="padding:11px 20px">📤 Compartilhar</button>
    <button class="btn" id="restartBtn" style="padding:11px 20px">↩ Jogar Novamente</button>
  </div>`;

  if(passed){
    const certData=pct===100?{rank:'Mestre CCNA',icon:'🏆',sub:'Desempenho Perfeito!',color1:'#F5C842',color2:'#BA7517',glow:'rgba(245,200,66,.35)'}:pct>=90?{rank:'Expert em Redes',icon:'🌟',sub:'Resultado excepcional!',color1:'#5EE0A8',color2:'#1D9E75',glow:'rgba(94,224,168,.3)'}:pct>=80?{rank:'CCNA Aprovado',icon:'✅',sub:'Pronto para o exame real!',color1:'#7AB8F5',color2:'#185FA5',glow:'rgba(122,184,245,.3)'}:{rank:'Técnico de Redes',icon:'📡',sub:'Aprovado no Limite!',color1:'#A89DF5',color2:'#534AB7',glow:'rgba(168,157,245,.3)'};
    const studentName=localStorage.getItem('ccna2_studentname')||'';
    const nameHtml=studentName?`<div class="cert-student-name">${studentName}</div>`:'<button class="btn" id="addNameBtn" style="font-size:11px;padding:4px 12px;margin-bottom:4px">✏️ Adicionar meu nome</button>';
    return`<div id="cert-screen">
      <div style="display:flex;justify-content:flex-end;gap:6px;margin-bottom:10px">
